cmake_minimum_required(VERSION 3.16)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(lucyn_search VERSION 0.1.0 LANGUAGES CXX)
  enable_testing()
endif()

add_library(lucyn_search STATIC src/search.cpp)
add_library(lucyn::search ALIAS lucyn_search)
target_include_directories(lucyn_search PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(lucyn_search PUBLIC cxx_std_17)
if(NOT MSVC)
  target_compile_options(lucyn_search PRIVATE -Wall -Wextra)
endif()

option(LUCYN_BUILD_TOOLS "Build module command-line tools" ON)
if(LUCYN_BUILD_TOOLS)
  add_executable(lucyn-search tools/lucyn-search.cpp)
  target_link_libraries(lucyn-search PRIVATE lucyn::search)
  if(NOT MSVC)
    target_compile_options(lucyn-search PRIVATE -Wall -Wextra)
  endif()
  if(CMAKE_RUNTIME_OUTPUT_DIRECTORY)
    set_target_properties(lucyn-search PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
  endif()
endif()
